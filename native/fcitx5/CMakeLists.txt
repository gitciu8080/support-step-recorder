cmake_minimum_required(VERSION 3.16)
project(ssr_text_bridge LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Fcitx5Core REQUIRED)
add_library(ssrtext MODULE ssrtext.cpp)
target_link_libraries(ssrtext PRIVATE Fcitx5::Core)
target_compile_features(ssrtext PRIVATE cxx_std_17)
install(TARGETS ssrtext LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
install(FILES ssrtext.conf DESTINATION share/fcitx5/addon)
